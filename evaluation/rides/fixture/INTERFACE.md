# Ride-hailing and food delivery platform: public interface

The package is `rides`, under `src/rides/`, and it exposes one class: `from rides import Platform`.
Everything else (modules, classes, storage layout) is up to you. Only the names below are fixed.

Conventions: identifiers are strings; money is `decimal.Decimal`; times are naive `datetime.datetime`
(UTC); dates are `datetime.date`; a location is a `(latitude, longitude)` tuple of floats.

```python
class Platform:
    # Lifecycle, clock and storage
    def __init__(self, start, db_path=None): ...
    @classmethod
    def open(cls, db_path) -> "Platform": ...
    def save(self) -> None: ...
    def now(self) -> datetime: ...
    def advance(self, minutes=0, seconds=0) -> None: ...

    # Accounts
    def create_admin(self, name, email, phone, password) -> str: ...
    def register_rider(self, name, email, phone, password, birth_date=None, referral_code=None) -> str: ...
    def register_driver(self, name, email, phone, password, license_number, license_expires) -> str: ...
    def register_courier(self, name, email, phone, password, vehicle) -> str: ...
    def register_restaurant(self, name, email, phone, password, location) -> str: ...
    def authenticate(self, email, password) -> str: ...
    def get_account(self, account_id) -> dict: ...   # id, role, name, email, phone, status
    def referral_code(self, rider_id) -> str: ...
    def suspend(self, admin_id, account_id, reason) -> None: ...
    def reactivate(self, admin_id, account_id) -> None: ...

    # Drivers, couriers and vehicles
    def add_vehicle(self, driver_id, plate, make, model, year, seats, category) -> str: ...
    def approve(self, admin_id, account_id) -> None: ...
    def go_online(self, account_id, location, vehicle_id=None) -> None: ...
    def go_offline(self, account_id) -> None: ...
    def update_location(self, account_id, location) -> None: ...
    def worker_status(self, account_id) -> dict: ...  # status, location, vehicle_id

    # Geography
    def add_zone(self, admin_id, name, center, radius_km, speed_kmh, airport=False) -> str: ...
    def distance_km(self, a, b) -> Decimal: ...
    def eta_minutes(self, a, b) -> int: ...
    def set_surge_cap(self, admin_id, zone_id, cap) -> None: ...

    # Rides
    def quote_ride(self, rider_id, pickup, dropoff, category) -> dict:
        ...  # quote_id, category, distance_km, duration_min, surge, fare, expires_at
    def request_ride(self, rider_id, quote_id, payment_method_id, promo_code=None, idempotency_key=None) -> str: ...
    def schedule_ride(self, rider_id, pickup, dropoff, category, pickup_at, payment_method_id) -> str: ...
    def accept_ride(self, driver_id, ride_id) -> None: ...
    def decline_ride(self, driver_id, ride_id) -> None: ...
    def driver_arrived(self, driver_id, ride_id) -> None: ...
    def start_ride(self, driver_id, ride_id) -> None: ...
    def complete_ride(self, driver_id, ride_id, route) -> None: ...
    def cancel_ride(self, actor_id, ride_id, reason="") -> None: ...
    def ride(self, ride_id) -> dict:
        ...  # id, rider_id, driver_id, status, category, quoted_fare, fare, distance_km, duration_min,
             # surge, cancellation_fee, wait_fee, discount, tip, charged, offered_to, pickup_at
    def add_tip(self, rider_id, ride_id, amount) -> None: ...
    def share_trip(self, rider_id, ride_id) -> str: ...
    def trip_status(self, token) -> dict: ...

    # Restaurants and menus
    def set_hours(self, restaurant_id, weekday, opens, closes) -> None: ...
    def is_open(self, restaurant_id) -> bool: ...
    def add_menu_item(self, restaurant_id, name, price, category, prep_minutes) -> str: ...
    def add_option_group(self, item_id, name, required, min_choices, max_choices) -> str: ...
    def add_option(self, group_id, name, price_delta) -> str: ...
    def set_item_available(self, restaurant_id, item_id, available) -> None: ...
    def menu(self, restaurant_id) -> list[dict]: ...

    # Food orders
    def place_order(self, rider_id, restaurant_id, lines, dropoff, payment_method_id,
                    promo_code=None, tip=Decimal("0"), idempotency_key=None) -> str: ...
    def accept_order(self, restaurant_id, order_id) -> None: ...
    def reject_order(self, restaurant_id, order_id, reason="") -> None: ...
    def mark_ready(self, restaurant_id, order_id) -> None: ...
    def pick_up(self, courier_id, order_id) -> None: ...
    def deliver(self, courier_id, order_id) -> None: ...
    def cancel_order(self, actor_id, order_id, reason="") -> None: ...
    def order(self, order_id) -> dict:
        ...  # id, rider_id, restaurant_id, courier_id, status, lines, subtotal, service_fee,
             # delivery_fee, small_order_fee, tax, discount, tip, total, charged, refunded, ready_at

    # Promotions
    def create_promo(self, admin_id, code, service, kind, value, max_discount=None, min_spend=Decimal("0"),
                     starts=None, ends=None, max_total_uses=None, max_uses_per_user=None,
                     first_order_only=False) -> None: ...

    # Payments and wallet
    def add_card(self, rider_id, number, exp_month, exp_year, cvc) -> str: ...
    def payment_methods(self, rider_id) -> list[dict]: ...  # id, kind, brand, last4
    def wallet_balance(self, rider_id) -> Decimal: ...
    def top_up_wallet(self, rider_id, card_id, amount) -> None: ...
    def payments(self, account_id) -> list[dict]: ...  # id, kind, ref, method_id, amount, at

    # Earnings, ratings, support, notifications, reports, audit
    def earnings(self, worker_id, start, end) -> dict: ...  # trips, fares, tips, fees, bonus, total
    def run_payouts(self, admin_id, week_start) -> dict: ...  # {worker_id: amount}
    def rate(self, actor_id, ref_id, stars, comment="", target="driver") -> None: ...
    def rating(self, account_id) -> Decimal | None: ...
    def report_issue(self, actor_id, ref_id, kind, text) -> str: ...
    def resolve_ticket(self, admin_id, ticket_id, refund_amount=Decimal("0")) -> None: ...
    def notifications(self, account_id) -> list[dict]: ...  # at, kind, ref
    def daily_report(self, admin_id, day) -> dict: ...
    def top_restaurants(self, admin_id, start, end, limit=5) -> list[dict]: ...
    def audit_log(self, admin_id) -> list[dict]: ...  # at, admin_id, action, target
```
