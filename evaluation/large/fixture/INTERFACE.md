# Online store backend: public interface

The package is `shop`, under `src/shop/`, and it exposes one class: `from shop import Shop`.
Everything else is up to you. Only the names below are fixed.

```python
class Shop:
    def add_product(self, sku, name, price, category, weight_kg) -> None: ...
    def get_product(self, sku) -> dict: ...          # keys: sku, name, price, category, weight_kg
    def list_products(self, category=None) -> list[dict]: ...
    def receive_stock(self, sku, quantity) -> None: ...
    def stock(self, sku) -> int: ...
    def add_coupon(self, code, percent, min_subtotal="0", expires=None, max_uses=None) -> None: ...
    def create_order(self, customer_email, items, coupon=None, on=None) -> str: ...  # items: {sku: quantity}
    def order(self, order_id) -> dict: ...           # keys: id, customer, status, lines, subtotal, discount,
                                                     # tax, shipping, total, paid, refunded
                                                     # each line: sku, quantity, unit_price, net, discount, tax
    def pay(self, order_id, amount, idempotency_key) -> str: ...
    def ship(self, order_id) -> None: ...
    def cancel(self, order_id) -> None: ...
    def refund(self, order_id, amount) -> None: ...
    def return_items(self, order_id, sku, quantity) -> None: ...
    def sales_report(self, start, end) -> dict: ...  # keys: orders, revenue, tax, units
    def save(self, path) -> None: ...
    @classmethod
    def load(cls, path) -> "Shop": ...
```

Monetary values are `decimal.Decimal`; dates are `datetime.date`.
